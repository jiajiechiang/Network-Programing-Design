Python Socket 網路通訊中階教學

突破單點限制：多執行緒、多工 I/O 與自訂通訊協定

中階 Socket 通訊挑戰：
在基礎篇中，我們的伺服器一次只能服務「一個」客戶端。在中階篇，我們將解決真實世界的網路難題：
1. 並行處理：讓伺服器同時服務多個客戶端 (Threading / Select)。
2. 資料邊界 (TCP 黏包)：TCP 是串流協定，我們需要自訂長度標頭來確保資料完整性。
3. 廣播與安全：實作 UDP 廣播機制以及 SSL/TLS 基礎加密。

Demo 1 超時設定與非阻塞 (Timeout)

實務上的網路連線常常會斷線或卡住。我們不能讓程式永遠等下去，必須設定 settimeout() 讓 Socket 具備超時中斷的能力。

tcp_timeout_client.py
import socket

HOST = '127.0.0.1'
PORT = 8888

with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
    # 設定超時時間為 3 秒。如果 3 秒內連線或接收不到資料，將引發 socket.timeout 例外
    s.settimeout(3.0)
    
    try:
        s.connect((HOST, PORT))
        s.sendall(b"Hello")
        
        # 若伺服器卡住不回應，3秒後會跳入 except 區塊
        data = s.recv(1024)
        print("收到資料：", data)
        
    except socket.timeout:
        print("錯誤：伺服器回應超時！請檢查網路狀態。")
    except ConnectionRefusedError:
        print("錯誤：連線被拒，伺服器可能未啟動。")

Demo 2 多執行緒伺服器 (Multithreaded Server)

為了解決伺服器一次只能服務一個客戶端的問題，每當有新客戶端連入 accept() 時，我們就開一個新的 執行緒 (Thread) 給他，主程式繼續等待下一個連線。

threaded_server.py
import socket
import threading

def handle_client(conn, addr):
    print(f"[新連線] {addr} 進入伺服器。")
    with conn:
        while True:
            data = conn.recv(1024)
            if not data:
                break
            # 回聲功能
            conn.sendall(data)
    print(f"[斷線] {addr} 已離開。")

def start_server():
    HOST, PORT = '127.0.0.1', 9000
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server:
        server.bind((HOST, PORT))
        server.listen()
        print(f"多執行緒伺服器啟動，監聽 {PORT}...")
        
        while True:
            # 接受新連線
            conn, addr = server.accept()
            # 建立並啟動新的執行緒來處理這個客戶端
            thread = threading.Thread(target=handle_client, args=(conn, addr))
            thread.start()
            print(f"[活躍連線數] {threading.active_count() - 1}")

if __name__ == "__main__":
    start_server()

Demo 3 I/O 多工處理：使用 select 打造聊天室

除了多執行緒，作業系統提供了更輕量的 select 模組，讓單一執行緒能夠「同時監視」多個 Socket 的狀態（誰傳資料來了就去讀誰），非常適合開發廣播型的群組聊天室。

select_chat_server.py
import socket
import select

HOST, PORT = '127.0.0.1', 9090
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind((HOST, PORT))
server.listen()

# 存放所有連線的 Socket，包含伺服器本身
sockets_list = [server]

print("Select 聊天室伺服器啟動...")
while True:
    # select 會卡住，直到 sockets_list 中有人發生讀取事件
    read_sockets, _, _ = select.select(sockets_list, [], [])
    
    for notified_socket in read_sockets:
        # 如果是伺服器本身有事件，代表有新客戶端連入
        if notified_socket == server:
            client_socket, client_addr = server.accept()
            sockets_list.append(client_socket)
            print(f"接受新連線：{client_addr}")
        
        # 如果是客戶端有事件，代表他傳送了訊息
        else:
            try:
                message = notified_socket.recv(1024)
                if not message:
                    sockets_list.remove(notified_socket)
                    continue
                
                # 將訊息廣播給其他所有人
                for client in sockets_list:
                    if client != server and client != notified_socket:
                        client.send(message)
            except:
                sockets_list.remove(notified_socket)

Demo 4 解決 TCP 黏包：自訂資料長度標頭 (struct)

TCP 是「資料流 (Stream)」，如果連續發送多筆資料，接收端可能會把兩筆資料黏在一起讀出來（黏包）。解決方案是在每筆資料前面加上 固定長度的標頭 (Header)，宣告接下來的資料有多長。

custom_protocol.py
import socket
import struct

def send_msg(sock, msg_bytes):
    # 使用 struct.pack 打包長度，'>I' 代表大端序無號整數 (固定 4 bytes)
    msg_length = len(msg_bytes)
    header = struct.pack('>I', msg_length)
    # 先傳標頭，再傳實際內容
    sock.sendall(header + msg_bytes)

def recv_msg(sock):
    # 先讀取固定的 4 bytes 標頭
    raw_msglen = recv_all(sock, 4)
    if not raw_msglen:
        return None
    
    # 解開標頭，取得實際資料長度
    msglen = struct.unpack('>I', raw_msglen)[0]
    # 依照長度讀取完整的內容
    return recv_all(sock, msglen)

def recv_all(sock, n):
    # 確保讀滿 n 個 bytes 才回傳
    data = bytearray()
    while len(data) < n:
        packet = sock.recv(n - len(data))
        if not packet:
            return None
        data.extend(packet)
    return data

Demo 5 UDP 廣播 (Broadcast)

想在同一個區域網路 (LAN) 內呼叫所有人？可以使用 UDP 廣播，將資料傳送到 <broadcast>，網域內所有監聽該 Port 的機器都能收到。

udp_broadcast.py
import socket
import time

# -- 廣播發送端 --
def broadcast_sender():
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
        # 開啟 Socket 的廣播權限
        s.setsockopt(socket.SOL_SOCKET, socket.SO_BROADCAST, 1)
        
        while True:
            msg = "【廣播】伺服器上線公告！"
            # 傳送到廣播位址與指定 Port
            s.sendto(msg.encode(), ('<broadcast>', 37020))
            print("已發送廣播...")
            time.sleep(5)

# -- 廣播接收端 --
def broadcast_receiver():
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
        # 允許重複使用 Port
        s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        # 綁定所有網卡 ('') 監聽該 Port
        s.bind(('', 37020))
        
        print("等待廣播訊息...")
        while True:
            data, addr = s.recvfrom(1024)
            print(f"收到來自 {addr} 的廣播：{data.decode()}")

Demo 6 SSL/TLS 安全加密通訊 (Secure Sockets)

明文傳輸非常危險。Python 內建的 ssl 模組可以幫原本的 Socket 加上一層防護罩，實作類似 HTTPS 的加密傳輸。

ssl_secure_client.py
import socket
import ssl

HOST = 'www.python.org'
PORT = 443  # HTTPS 預設 Port

# 1. 建立預設的安全上下文環境
context = ssl.create_default_context()

with socket.create_connection((HOST, PORT)) as raw_socket:
    # 2. 將普通的 Socket 包裝成 SSL Socket
    with context.wrap_socket(raw_socket, server_hostname=HOST) as secure_socket:
        
        print(f"加密連線成功！使用的協定：", secure_socket.version())
        
        # 透過加密通道發送 HTTP 請求
        request = f"GET / HTTP/1.1\r\nHost: {HOST}\r\nConnection: close\r\n\r\n"
        secure_socket.sendall(request.encode('utf-8'))
        
        # 接收安全的回應資料
        response = secure_socket.recv(1024)
        print(response.decode('utf-8')[:200]) # 僅印出前200字元
