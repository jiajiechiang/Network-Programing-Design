Demo 1 高效能非阻塞 I/O：使用 selectors 模組

在 Linux C 語言中常用的 `epoll` 或 Windows 的 `IOCP` 是最高效能的 I/O 模型。Python 提供了 selectors 模組，會自動根據你的作業系統選擇最佳的底層機制，比傳統的 select 更強大、更具擴展性。

advanced_selectors_server.py
import socket
import selectors

# 建立最佳的 selector 物件 (Linux: epoll, Mac: kqueue, Win: select)
sel = selectors.DefaultSelector()

def accept_wrapper(sock):
    conn, addr = sock.accept()
    print(f"[連線] 接受來自 {addr} 的連線")
    conn.setblocking(False) # 設定為非阻塞模式
    # 註冊客戶端 Socket，準備讀取 (EVENT_READ)
    sel.register(conn, selectors.EVENT_READ, data=read_wrapper)

def read_wrapper(conn):
    try:
        data = conn.recv(1024)
        if data:
            print(f"收到訊息: {data.decode()}")
            conn.sendall(data) # Echo 回傳
        else:
            print("[斷線] 客戶端關閉連線")
            sel.unregister(conn)
            conn.close()
    except ConnectionResetError:
        sel.unregister(conn)
        conn.close()

HOST, PORT = '127.0.0.1', 9999
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind((HOST, PORT))
server.listen()
server.setblocking(False)

# 註冊伺服器 Socket，當有新連線時呼叫 accept_wrapper
sel.register(server, selectors.EVENT_READ, data=accept_wrapper)

print(f"Selectors 伺服器啟動於 {HOST}:{PORT}")
try:
    while True:
        # 阻塞等待，直到有已註冊的 Socket 發生事件
        events = sel.select(timeout=None)
        for key, mask in events:
            callback = key.data # 取得綁定的函式
            callback(key.fileobj)
finally:
    sel.close()

Demo 2 現代化並行：Asyncio Socket 伺服器

Python 3 引入了強大的 asyncio。我們不需手動處理繁瑣的 selectors，使用 async/await 語法就能寫出極高併發 (C10K problem) 的非阻塞網路伺服器。

asyncio_tcp_server.py
import asyncio

async def handle_client(reader, writer):
    addr = writer.get_extra_info('peername')
    print(f"[連線] {addr} 已連入")
    
    while True:
        # 非同步讀取資料，不會卡住主執行緒
        data = await reader.read(100)
        if not data:
            break
            
        message = data.decode()
        print(f"收到 {addr} 訊息: {message!r}")
        
        # 非同步寫入回覆
        writer.write(data)
        await writer.drain() # 確保緩衝區寫入完畢
        
    print(f"[斷線] {addr} 已離開")
    writer.close()
    await writer.wait_closed()

async def main():
    # 使用 asyncio 建立底層 Socket 伺服器
    server = await asyncio.start_server(
        handle_client, '127.0.0.1', 8888
    )
    
    addrs = ', '.join(str(sock.getsockname()) for sock in server.sockets)
    print(f"Asyncio 伺服器運行於 {addrs}")
    
    async with server:
        await server.serve_forever()

if __name__ == "__main__":
    asyncio.run(main())

Demo 3 從零打造 HTTP Server：手刻網頁伺服器

HTTP 協定本質上就是建立在 TCP 之上的「純文字通訊」。我們可以透過 Socket 接收瀏覽器的請求，並按照 HTTP 規範回傳標頭 (Headers) 與 HTML 內容，讓瀏覽器成功顯示網頁！

custom_http_server.py
import socket

HOST, PORT = '127.0.0.1', 8080

with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server:
    server.bind((HOST, PORT))
    server.listen(1)
    print(f"HTTP 伺服器啟動！請在瀏覽器輸入 http://{HOST}:{PORT}")
    
    while True:
        conn, addr = server.accept()
        with conn:
            request_data = conn.recv(1024).decode('utf-8')
            if not request_data:
                continue
                
            # 印出瀏覽器發送的 HTTP Request Header
            print("收到請求：", request_data.split('\n')[0])
            
            # 準備回傳的 HTTP Response (狀態碼 200 OK)
            http_response = (
                "HTTP/1.1 200 OK\r\n"
                "Content-Type: text/html; charset=utf-8\r\n"
                "\r\n"  # 空行代表 Header 結束，接著是 Body
                "<html><head><title>Python Socket</title></head>"
                "<body><h1 style='color:green;'>成功了！</h1>"
                "<p>這是我用純 Socket 手刻的 HTTP 伺服器！</p></body></html>"
            )
            
            # 將文字編碼為 bytes 並送給瀏覽器
            conn.sendall(http_response.encode('utf-8'))

Demo 4 UDP 群播 (Multicast)：指定群組廣播

普通的 UDP 廣播會吵到網域內的所有人。群播 (Multicast) 允許我們將資料發送到特定的「D 類 IP 位址 (例如 224.1.1.1)」，只有主動加入該群組的 Socket 才會收到資料，常見於串流媒體與 IoT 設備發現。

udp_multicast.py
import socket
import struct

MCAST_GRP = '224.1.1.1'
MCAST_PORT = 5007

# == 接收端 (Receiver) ==
def start_receiver():
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
    sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    sock.bind(('', MCAST_PORT))
    
    # 告訴作業系統將這個 Socket 加入特定的群播群組 (IGMP)
    mreq = struct.pack("4sl", socket.inet_aton(MCAST_GRP), socket.INADDR_ANY)
    sock.setsockopt(socket.IPPROTO_IP, socket.IP_ADD_MEMBERSHIP, mreq)
    
    print("等待群播訊息...")
    while True:
        print(sock.recv(1024).decode())

# == 發送端 (Sender) ==
def start_sender():
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
    # 設定 TTL (Time to Live)，決定封包能跨越多少個路由器 (1 代表僅限本地子網)
    sock.setsockopt(socket.IPPROTO_IP, socket.IP_MULTICAST_TTL, 1)
    
    msg = "【群播】給 224.1.1.1 頻道訂閱者的專屬訊息！"
    sock.sendto(msg.encode(), (MCAST_GRP, MCAST_PORT))
    print("訊息已發送。")

Demo 5 Unix Domain Sockets：本機極速通訊 (IPC)

如果 Server 和 Client 都在「同一台機器」上（例如 Nginx 與 uWSGI），走 TCP/IP 協定繞一圈網路層太慢了。在 Linux/macOS 系統中，我們可以使用 AF_UNIX 透過本機檔案系統進行極速的行程間通訊 (IPC)。

unix_domain_socket.py (僅限 Linux / macOS)
import socket
import os

SOCKET_FILE = '/tmp/my_ipc.sock'

# 清理遺留的 Socket 檔案
if os.path.exists(SOCKET_FILE):
    os.remove(SOCKET_FILE)

# 使用 AF_UNIX 建立 IPC Socket
with socket.socket(socket.AF_UNIX, socket.SOCK_STREAM) as server:
    server.bind(SOCKET_FILE)
    server.listen()
    print(f"UDS 伺服器監聽本機檔案：{SOCKET_FILE}")
    
    # 模擬客戶端連線
    def client_task():
        with socket.socket(socket.AF_UNIX, socket.SOCK_STREAM) as client:
            client.connect(SOCKET_FILE)
            client.sendall(b"Hello via local filesystem!")
            
    import threading
    threading.Thread(target=client_task).start()
    
    # 伺服器接收
    conn, _ = server.accept()
    with conn:
        data = conn.recv(1024)
        print("UDS 伺服器收到：", data.decode())

Demo 6 Raw Sockets：底層封包側錄 (Packet Sniffing)

駭客或網管軟體（如 Wireshark）是如何看到別人的封包的？他們使用了 Raw Sockets，繞過 TCP/UDP 層，直接抓取 IP 層甚至乙太網路層的原始封包。（注意：此程式需在 Linux 下以 root 管理員權限執行）。
